try { UrlFetchApp.fetch(SUPABASE_URL + "/rest/v1/pelanggan", options); Logger.log("Data berhasil dikirim ke Supabase: " + payload.nama_pelanggan); } catch (error) { Logger.log("Error sync ke Supabase: " + error.toString());
